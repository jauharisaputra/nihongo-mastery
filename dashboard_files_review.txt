    <script>

        const USER_ID =
            localStorage.getItem('userId') || '3';

        const USER_NAME =
            localStorage.getItem('userName') || 'Premium User';

        const USER_ROLE =
            localStorage.getItem('userRole') || 'premium';

        const GAS_URL =
            'https://script.google.com/macros/s/AKfycbzdDuFwNNMQQUoV3YAhzJctS9M2t0wr8c3WDwIU5fc1q8Lb1mHmFAwsCUEXIL_s1zMC/exec';

        document.getElementById(
            'userNameDash'
        ).textContent =
            USER_NAME;

        document.getElementById(
            'userRoleBadge'
        ).textContent =
            USER_ROLE.toUpperCase();

        debug('USER ID: ' + USER_ID);

        loadDashboard();

        function debug(msg) {

            console.log(msg);

            document.getElementById(
                'debugBox'
            ).innerHTML += `<br>${msg}`;
        }

        async function loadDashboard() {

            try {

                const url =
                    `${GAS_URL}?action=results&user_id=${USER_ID}`;

                debug('FETCH: ' + url);

                const response =
                    await fetch(url);

                const text =
                    await response.text();

                debug('RAW RESPONSE: ' + text);

                const data =
                    JSON.parse(text);

                updateResults(
                    data.results || []
                );

            } catch (err) {

                debug(err.message);

            }

        }

        function updateResults(results) {

            const tbody =
                document.getElementById(
                    'resultsTbody'
                );

            if (!results.length) {

                tbody.innerHTML = `
                    <tr>
                        <td colspan="3">
                            Belum ada hasil ujian
                        </td>
                    </tr>
                `;

                return;
            }

            tbody.innerHTML = '';

            let bestScore = 0;
            let bestExam = '';

            results.forEach(result => {

                const score =
                    Number(result.score || 0);

                if (score > bestScore) {

                    bestScore = score;
                    bestExam =
                        result.exam_path || '';

                }

                tbody.innerHTML += `
                    <tr>

                        <td>
                            ${result.exam_path || '-'}
                        </td>

                        <td>
                            ${score}%
                        </td>

                        <td>
                            ${result.date || '-'}
                        </td>

                    </tr>
                `;
            });

            document.getElementById(
                'totalExams'
            ).textContent =
                results.length;

            document.getElementById(
                'bestScore'
            ).textContent =
                bestScore + '%';

            document.getElementById(
                'progressText'
            ).textContent =
                bestScore + '%';

            document.getElementById(
                'progressFill'
            ).style.width =
                bestScore + '%';

            // CERTIFICATE
            if (bestScore >= 80) {

                document.getElementById(
                    'certSection'
                ).style.display =
                    'block';

                document.getElementById(
                    'certName'
                ).textContent =
                    USER_NAME;

                document.getElementById(
                    'certScore'
                ).textContent =
                    `${bestScore}%`;

                document.getElementById(
                    'certDesc'
                ).innerHTML =
                    `
                    Telah berhasil menyelesaikan
                    <strong>${bestExam}</strong>
                    dengan nilai terbaik dan memenuhi standar kelulusan Nihongo Mastery Premium.
                    `;
            }

        }

        function startNewExam() {

            const examPath =
                document.getElementById(
                    'examSelect'
                ).value;

            window.location.href =
                `exam.html?path=${encodeURIComponent(examPath)}&user=${USER_ID}`;
        }

   async function downloadCertificate() {

        const button =
            document.getElementById(
                'downloadBtn'
            );

        const certificate =
            document.getElementById(
                'certificateCard'
            );

        try {

            // =====================================
            // HIDE BUTTON
            // =====================================

            button.style.display = 'none';

            // =====================================
            // WAIT RENDER
            // =====================================

            await new Promise(resolve =>
                setTimeout(resolve, 300)
            );

            // =====================================
            // CANVAS
            // =====================================

            const canvas =
                await html2canvas(certificate, {

                    scale: 2,

                    useCORS: true,

                    backgroundColor: '#ffffff'

                });

            const imgData =
                canvas.toDataURL(
                    'image/jpeg',
                    1.0
                );

            // =====================================
            // PDF
            // =====================================

            const {
                jsPDF
            } = window.jspdf;

            const pdf =
                new jsPDF({

                    orientation: 'landscape',

                    unit: 'mm',

                    format: 'a4'

                });

            const pdfWidth =
                297;

            const pdfHeight =
                210;

            pdf.addImage(

                imgData,

                'JPEG',

                0,

                0,

                pdfWidth,

                pdfHeight

            );

            // =====================================
            // SAVE
            // =====================================

            pdf.save(
                `certificate-${USER_NAME}.pdf`
            );

        } catch (err) {

            console.error(err);

            alert(
                'Gagal membuat PDF'
            );

        } finally {

            // =====================================
            // SHOW BUTTON AGAIN
            // =====================================

            button.style.display =
                'inline-block';
        }
    }
        function logout() {

            localStorage.clear();

            window.location.href =
                'index.html';
        }

    </script>

        }

        return data;

    } catch (err) {

        throw err;
    }
}

        Date.now() + (10 * 60 * 1000);

function updateCountdownDisplay() {

    const timer =
        document.getElementById('countdown-timer');

    if (!timer) return;

    const now = Date.now();

    const diff =
        promoEndTime - now;

    if (diff <= 0) {

        timer.textContent = '00:00';

        return;
    }

    const minutes =
        Math.floor(diff / 60000);

    const seconds =
        Math.floor((diff % 60000) / 1000);

    timer.textContent =
        `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
}

// JWT PARSER
// ========================================
function parseJwt(token) {

    try {

        const base64Url =
            token.split('.')[1];

        const base64 =
            base64Url
            .replace(/-/g, '+')
            .replace(/_/g, '/');

        const jsonPayload =
            decodeURIComponent(
                atob(base64)
                .split('')
                .map(c =>
                    '%' +
                    ('00' + c.charCodeAt(0).toString(16)).slice(-2)
                )
                .join('')
            );

        return JSON.parse(jsonPayload);

    } catch (err) {

        console.error(err);

        return null;
    }
}

async function restoreSession() {

    const userId =
        localStorage.getItem('userId');

    const email =
        localStorage.getItem('userEmail');

    if (!userId || !email) {

        console.log('⚠️ NO SESSION');

        return;
    }

    try {

        const response =
            await fetch(
                `${GAS_URL}?action=check_user&email=${encodeURIComponent(email)}`
            );

        const data =
            await response.json();

        console.log('RESTORE:', data);

        if (data.success) {

            const restoredUser = {

                user_id: data.user_id,

                email: data.email,

                name: data.name,

                role: data.role
            };

            updateUserUI(restoredUser);

            console.log('✅ SESSION RESTORED');
        }

    } catch (err) {

        console.error(err);
    }
}

function handleRequest(e) {

      e.postData.contents
    ) {

        const jsonData =
          JSON.parse(e.postData.contents);

        data = {
          ...data,
          ...jsonData
        };

      } catch (err) {

        console.log(
          'JSON parse failed'
        );

      }

    }

    console.log(
      'REQUEST:',
      JSON.stringify(data)
    );

    const action =
      String(
        data.action ||
        data.method ||
        ''
      ).trim();

      action
    );

      SpreadsheetApp.openById(

    // ROUTING

          time: new Date()

        return registerUser(

// REGISTER USER

function registerUser(
  ss,
  data
) {

  const userSheet =
    ss.getSheetByName(
      'users'
    );

  if (!userSheet) {

    return jsonResponse({

      success: false,
      error:
        'Sheet users not found'

    });

  }

  const users =
    userSheet
    .getDataRange()
    .getValues();

  const email =
    String(data.email || '')
    .trim()
    .toLowerCase();

  const name =
    String(data.name || 'User');

  // ========================================
  // VALIDATE EMAIL
  // ========================================
  if (
    !email ||
    !email.includes('@')
  ) {

    return jsonResponse({

      success: false,
      error: 'Invalid email'

    });

  }

  // ========================================
  // CHECK EXISTING USER
  // ========================================
  for (
    let i = 1;
    i < users.length;
    i++
  ) {

    const existingEmail =
      String(users[i][1] || '')
      .trim()
      .toLowerCase();

    if (
      existingEmail === email
    ) {

      return jsonResponse({

        success: true,
        existing: true,

        user_id:
          users[i][0],

        email:
          users[i][1],

        name:
          users[i][2],

        role:
          users[i][3] ||
          'student'

      });

    }

  }

  // ========================================
  // CREATE USER
  // ========================================
  const userId =
    Utilities
    .getUuid()
    .slice(0, 8);

  userSheet.appendRow([

    userId,
    email,
    name,
    'student',
    new Date()

  ]);

  return jsonResponse({

    success: true,
    existing: false,

    user_id:
      userId,

    email:
      email,

    name:
      name,

    role:
      'student'

  });

}

// GET RESULTS
// ========================================
function getResults(
  ss,
  userId
) {

  const resultSheet =
    ss.getSheetByName(
      'results'
    );

  if (!resultSheet) {

    return jsonResponse({

      success: false,
      error:
        'Sheet results not found'

    });

  }

  const rows =
    resultSheet
    .getDataRange()
    .getValues();

  let results = [];

  for (
    let i = 1;
    i < rows.length;
    i++
  ) {

    const rowUserId =
      String(rows[i][1] || '');

    if (
      rowUserId ===
      String(userId)
    ) {

      results.push({

        id:
          rows[i][0],

        user_id:
          rows[i][1],

        exam_path:
          rows[i][2],

        score:
          rows[i][3],

        total_questions:
          rows[i][4],

        time_used:
          rows[i][5],

        date:
          formatDate(
            rows[i][6]
          )

      });

    }

  }

  // ========================================
  // SORT TERBARU
  // ========================================
  results.reverse();

  return jsonResponse({

    success: true,
    total: results.length,
    results: results

  });

}

function submitPayment(

    }

      Number(
        data.amount || 0
      );

    if (amount <= 0) {

      return jsonResponse({

        success: false,
        error:
          'Invalid amount'

      });

    }

    const paymentId =
      Utilities

      .slice(0, 8);

    paymentSheet.appendRow([

      data.user_id || '',

      data.plan || 'premium',

      data.order_id || '',

      'pending',

      data.proof || '',

      new Date()

    ]);

    return jsonResponse({

      success: true,
      payment_id: paymentId,
      status: 'pending'

    });

  } finally {

    lock.releaseLock();

  }

}

// APPROVE PAYMENT
// ========================================
function approvePayment(
  ss,
  data
) {

  // ========================================
  // ADMIN SECURITY
  // ========================================
  if (
    data.admin_key !==
    ADMIN_KEY
  ) {

    return jsonResponse({

      success: false,
      error: 'Unauthorized'

    });

  }

  const paymentSheet =
    ss.getSheetByName(
      'payments'
    );

  const userSheet =
    ss.getSheetByName(
      'users'
    );

  if (
    !paymentSheet ||
    !userSheet
  ) {

    return jsonResponse({

      success: false,
      error:
        'Sheet not found'

    });

  }

  const payments =
    paymentSheet
    .getDataRange()
    .getValues();

  const orderId =
    data.order_id || '';

  let userId = '';
  let plan = 'premium';

  // ========================================
  // UPDATE PAYMENT
  // ========================================
  for (
    let i = 1;
    i < payments.length;
    i++
  ) {

    if (
      payments[i][4] ==
      orderId
    ) {

      paymentSheet
        .getRange(i + 1, 6)
        .setValue(
          'approved'
        );

      userId =
        payments[i][1];

      plan =
        payments[i][2];

      break;

    }

  }

  if (!userId) {

    return jsonResponse({

      success: false,
      error:
        'Payment not found'

    });

  }

  // ========================================
  // UPDATE USER ROLE
  // ========================================
  const users =
    userSheet
    .getDataRange()
    .getValues();

  for (
    let i = 1;
    i < users.length;
    i++
  ) {

    if (
      users[i][0] ==
      userId
    ) {

      userSheet
        .getRange(i + 1, 4)
        .setValue(plan);

      break;

    }

  }

  return jsonResponse({

    success: true,
    user_id: userId,
    role: plan,
    order_id: orderId

  });

}

                const response =
                    await fetch(GAS_URL, {

                        method: 'POST',

                        headers: {
                            'Content-Type': 'text/plain;charset=utf-8'
                        },

                        body: JSON.stringify({

                            action: 'submit_payment',

                            order_id:
                                document.getElementById('orderId').value,

                            user_id:
                                document.getElementById('userId').value,

                            plan:
                                document.getElementById('plan').value,

                            amount:
                                document.getElementById('amount').value,

                            proof:
                                base64Image
                        })
                    });

                const data =
                    await response.json();

                console.log('✅ RESPONSE:', data);

                loading.style.display =
                    'none';

                if (data.success) {

                    document.getElementById('successBox')
                        .style.display = 'block';

                    setTimeout(() => {

                        alert(
                            'Bukti pembayaran berhasil dikirim!\n\nAdmin akan memverifikasi pembayaran Anda.'
                        );

                        window.location.href =
                            'index.html';

                    }, 1000);

                } else {

                    btn.disabled = false;

                    alert(
                        'Upload gagal:\n' +
                        data.error
                    );
                }

            } catch (err) {

                console.error(err);

                document.getElementById('loadingText')
                    .style.display = 'none';

                document.getElementById('uploadBtn')
                    .disabled = false;

                alert(
                    'Upload gagal!\nPeriksa koneksi atau Apps Script.'
                );
            }
        }
